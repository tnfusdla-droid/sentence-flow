  correctCount: number;
  lastStudiedAt?: number;

interface StudyRecord {
  id: string;
  sentenceId: string;
  mode: 'word-arrange' | 'typing';
  userAnswer: string;
  isCorrect: boolean;
  attemptCount: number;
  studiedAt: number;
  duration: number; // in seconds
}

  id: string;
  mode: StudyMode;
  totalQuestions: number;
  correctCount: number;
  incorrectCount: number;
  firstTryCorrectCount: number;
  startedAt: number;
  completedAt: number;
  incorrectSentenceIds: string[];
}

interface WordToken {
  id: string;
  text: string;
}

// ==========================================
// UTILS & ANSWER NORMALIZATION
// ==========================================
function normalizeString(str: string): string {
  return str
    .trim()
    .toLowerCase()
    .replace(/\s+/g, ' ');
}

function tokenizeSentence(text: string): WordToken[] {
  // Split by whitespace while preserving punctuation attached to words
  const rawWords = text.trim().split(/\s+/);
  return rawWords.map((word, index) => ({
    id: `${index}-${word}-${Math.random().toString(36).substring(2, 6)}`,
    text: word
  }));
}

function compareDirectAnswers(userAns: string, correctAns: string): boolean {
  const normUser = normalizeString(userAns);
  const normCorrect = normalizeString(correctAns);
  return normUser === normCorrect;
}

// ==========================================
// MAIN COMPONENT
// ==========================================
export default function App() {
  // Navigation & UI States
  const [currentView, setCurrentView] = useState<View>('dashboard');
  const [activeModal, setActiveModal] = useState<'delete' | 'reset' | 'info' | null>(null);
  const [modalData, setModalData] = useState<any>(null);
  const [toastMessage, setToastMessage] = useState<string | null>(null);

  // Core Data States
  const [sentences, setSentences] = useState<Sentence[]>([]);
  const [records, setRecords] = useState<StudyRecord[]>([]);
  const [sessions, setSessions] = useState<StudySession[]>([]);

  // Editing / Form State
  const [editingSentenceId, setEditingSentenceId] = useState<string | null>(null);

  // Study Setup State
  const [studyConfig, setStudyConfig] = useState({
    mode: 'word-arrange' as StudyMode,
    scope: 'all' as 'all' | 'favorites' | 'unstudied' | 'incorrect' | 'category' | 'custom',
    categoryFilter: '',
    questionCount: 10,
    order: 'random' as QuestionOrder,
    customSelectedIds: [] as string[]
  });

  // Active Study Session State
  const [activeSessionQueue, setActiveSessionQueue] = useState<Sentence[]>([]);
  const [currentIndex, setCurrentIndex] = useState(0);
  const [sessionStartTime, setSessionStartTime] = useState(0);
  const [currentQuestionAttempts, setCurrentQuestionAttempts] = useState(0);
  const [currentQuestionStartTime, setCurrentQuestionStartTime] = useState(0);
  
  // Mode A specific state
  const [poolWords, setPoolWords] = useState<WordToken[]>([]);
  const [assembledWords, setAssembledWords] = useState<WordToken[]>([]);
  const [wordCheckResult, setWordCheckResult] = useState<'idle' | 'correct' | 'incorrect'>('idle');

  // Mode B specific state
  const [typedAnswer, setTypedAnswer] = useState('');
  const [typingCheckResult, setTypingCheckResult] = useState<'idle' | 'correct' | 'incorrect'>('idle');
  const [showAnswerInTyping, setShowAnswerInTyping] = useState(false);

  // Session Results State
  const [lastCompletedSession, setLastCompletedSession] = useState<StudySession | null>(null);

  // Filter & Search state for Sentence List
  const [searchQuery, setSearchQuery] = useState('');
  const [filterCategory, setFilterCategory] = useState('all');
  const [filterDifficulty, setFilterDifficulty] = useState('all');
  const [filterStatus, setFilterStatus] = useState('all'); // all, favorite, studied, unstudied
  const [sortBy, setSortBy] = useState<'latest' | 'oldest' | 'alphabetical'>('latest');

  // ==========================================
  // LOCALSTORAGE INITIALIZATION
  // ==========================================
  useEffect(() => {
    try {
      const savedSentences = localStorage.getItem(SENTENCES_KEY);
      const savedRecords = localStorage.getItem(RECORDS_KEY);
      const savedSessions = localStorage.getItem(SESSIONS_KEY);

      if (savedSentences) {
        setSentences(JSON.parse(savedSentences));
      } else {
        // Initialize with samples
        const initial: Sentence[] = SAMPLE_SENTENCES.map((s, idx) => ({
          ...s,
          id: `sample-${idx + 1}-${Date.now()}`,
          createdAt: Date.now() - idx * 100000,
          updatedAt: Date.now() - idx * 100000,
          studyCount: 0,
          correctCount: 0
        }));
        setSentences(initial);
        localStorage.setItem(SENTENCES_KEY, JSON.stringify(initial));
      }

      if (savedRecords) setRecords(JSON.parse(savedRecords));
      if (savedSessions) setSessions(JSON.parse(savedSessions));
    } catch (e) {
      console.error("Failed to load data from localStorage", e);
      showToast("로컬 데이터 로드 중 오류가 발생했습니다.");
    }
  }, []);

  // Sync data to localStorage
  useEffect(() => {
    if (sentences.length > 0) {
      localStorage.setItem(SENTENCES_KEY, JSON.stringify(sentences));
    }
  }, [sentences]);

  useEffect(() => {
    localStorage.setItem(RECORDS_KEY, JSON.stringify(records));
  }, [records]);

  useEffect(() => {
    localStorage.setItem(SESSIONS_KEY, JSON.stringify(sessions));
  }, [sessions]);

  // Toast helper
  const showToast = (msg: string) => {
    setToastMessage(msg);
    setTimeout(() => {
      setToastMessage(null);
    }, 3500);
  };

  // ==========================================
  // SENTENCE CRUD OPERATIONS
  // ==========================================
  const handleSaveSentence = (data: {
    englishText: string;
    koreanMeaning: string;
    category: string;
    difficulty: Difficulty;
    memo: string;
    isFavorite: boolean;
  }) => {
    if (!data.englishText.trim()) {
      showToast("영어 문장을 입력해주세요.");
      return;
    }

    if (editingSentenceId) {
      // Edit
      setSentences(prev => prev.map(s => s.id === editingSentenceId ? {
        ...s,
        ...data,
        updatedAt: Date.now()
      } : s));
      showToast("문장이 성공적으로 수정되었습니다.");
      setEditingSentenceId(null);
    } else {
      // Create
      const newSentence: Sentence = {
        id: `sent-${Date.now()}-${Math.random().toString(36).substring(2, 6)}`,
        ...data,
        createdAt: Date.now(),
        updatedAt: Date.now(),
        studyCount: 0,
        correctCount: 0
      };
      setSentences(prev => [newSentence, ...prev]);
      showToast("새로운 문장이 등록되었습니다!");
    }
    setCurrentView('list');
  };

  const handleDeleteSentence = (id: string) => {
    setSentences(prev => prev.filter(s => s.id !== id));
    showToast("문장이 삭제되었습니다.");
    setActiveModal(null);
  };

  const handleToggleFavorite = (id: string, e?: React.MouseEvent) => {
    if (e) e.stopPropagation();
    setSentences(prev => prev.map(s => s.id === id ? { ...s, isFavorite: !s.isFavorite } : s));
  };

  // ==========================================
  // STUDY SESSION LOGIC
  // ==========================================
  const startStudySession = () => {
    let targetPool = [...sentences];

    // Filter by scope
    if (studyConfig.scope === 'favorites') {
      targetPool = targetPool.filter(s => s.isFavorite);
    } else if (studyConfig.scope === 'unstudied') {
      targetPool = targetPool.filter(s => s.studyCount === 0);
    } else if (studyConfig.scope === 'incorrect') {
      const incorrectIds = new Set(records.filter(r => !r.isCorrect).map(r => r.sentenceId));
      targetPool = targetPool.filter(s => incorrectIds.has(s.id));
    } else if (studyConfig.scope === 'category' && studyConfig.categoryFilter) {
      targetPool = targetPool.filter(s => s.category === studyConfig.categoryFilter);
    } else if (studyConfig.scope === 'custom') {
      targetPool = targetPool.filter(s => studyConfig.customSelectedIds.includes(s.id));
    }

    if (targetPool.length === 0) {
      showToast("조건에 해당하는 문장이 없습니다. 다른 조건을 선택해주세요.");
      return;
    }

    // Order
    if (studyConfig.order === 'random') {
      targetPool.sort(() => Math.random() - 0.5);
    } else if (studyConfig.order === 'incorrect-first') {
      const incorrectIds = new Set(records.filter(r => !r.isCorrect).map(r => r.sentenceId));
      targetPool.sort((a, b) => {
        const aInc = incorrectIds.has(a.id) ? 0 : 1;
        const bInc = incorrectIds.has(b.id) ? 0 : 1;
        return aInc - bInc;
      });
    }

    // Limit count
    if (studyConfig.questionCount > 0 && studyConfig.questionCount < targetPool.length) {
      targetPool = targetPool.slice(0, studyConfig.questionCount);
    }

    setActiveSessionQueue(targetPool);
    setCurrentIndex(0);
    setSessionStartTime(Date.now());
    
    // Init first question
    initQuestion(targetPool[0], studyConfig.mode);
    setCurrentView('study');
  };

  const initQuestion = (sentence: Sentence, mode: StudyMode) => {
    setCurrentQuestionAttempts(0);
    setCurrentQuestionStartTime(Date.now());
    setWordCheckResult('idle');
    setTypingCheckResult('idle');
    setTypedAnswer('');
    setShowAnswerInTyping(false);

    const actualMode = mode === 'mixed' 
      ? (Math.random() > 0.5 ? 'word-arrange' : 'typing') 
      : mode;

    if (actualMode === 'word-arrange' || mode === 'word-arrange' || (mode === 'mixed' && actualMode === 'word-arrange')) {
      const tokens = tokenizeSentence(sentence.englishText);
      // Shuffle tokens for pool
      const shuffled = [...tokens].sort(() => Math.random() - 0.5);
      setPoolWords(shuffled);
      setAssembledWords([]);
    }
  };

  // Word Arrange mode handlers
  const handleSelectPoolWord = (token: WordToken) => {
    setPoolWords(prev => prev.filter(w => w.id !== token.id));
    setAssembledWords(prev => [...prev, token]);
  };

  const handleReturnAssembledWord = (token: WordToken) => {
    setAssembledWords(prev => prev.filter(w => w.id !== token.id));
    setPoolWords(prev => [...prev, token]);
  };

  const handleResetWordArrange = () => {
    const currentSentence = activeSessionQueue[currentIndex];
    if (!currentSentence) return;
    const tokens = tokenizeSentence(currentSentence.englishText);
    setPoolWords([...tokens].sort(() => Math.random() - 0.5));
    setAssembledWords([]);
    setWordCheckResult('idle');
  };

  const handleCheckWordArrange = () => {
    const currentSentence = activeSessionQueue[currentIndex];
    const correctTokens = tokenizeSentence(currentSentence.englishText);

    if (assembledWords.length !== correctTokens.length) {
      showToast("모든 단어를 배열해주세요.");
      return;
    }

    const userString = assembledWords.map(w => w.text).join(' ');
    const correctString = correctTokens.map(w => w.text).join(' ');
    const isCorrect = userString === correctString;

    const attempts = currentQuestionAttempts + 1;
    setCurrentQuestionAttempts(attempts);
    setWordCheckResult(isCorrect ? 'correct' : 'incorrect');

    // Record study
    const duration = Math.round((Date.now() - currentQuestionStartTime) / 1000);
    const newRecord: StudyRecord = {
      id: `rec-${Date.now()}`,
      sentenceId: currentSentence.id,
      mode: 'word-arrange',
      userAnswer: userString,
      isCorrect,
      attemptCount: attempts,
      studiedAt: Date.now(),
      duration
    };
    setRecords(prev => [newRecord, ...prev]);

    // Update sentence stats
    setSentences(prev => prev.map(s => {
      if (s.id === currentSentence.id) {
        return {
          ...s,
          studyCount: s.studyCount + 1,
          correctCount: isCorrect ? s.correctCount + 1 : s.correctCount,
          lastStudiedAt: Date.now()
        };
      }
      return s;
    }));
  };

  // Direct Typing mode handlers
  const handleCheckTyping = () => {
    const currentSentence = activeSessionQueue[currentIndex];
    const isCorrect = compareDirectAnswers(typedAnswer, currentSentence.englishText);

    const attempts = currentQuestionAttempts + 1;
    setCurrentQuestionAttempts(attempts);
    setTypingCheckResult(isCorrect ? 'correct' : 'incorrect');

    const duration = Math.round((Date.now() - currentQuestionStartTime) / 1000);
    const newRecord: StudyRecord = {
      id: `rec-${Date.now()}`,
      sentenceId: currentSentence.id,
      mode: 'typing',
      userAnswer: typedAnswer,
      isCorrect,
      attemptCount: attempts,
      studiedAt: Date.now(),
      duration
    };
    setRecords(prev => [newRecord, ...prev]);

    setSentences(prev => prev.map(s => {
      if (s.id === currentSentence.id) {
        return {
          ...s,
          studyCount: s.studyCount + 1,
          correctCount: isCorrect ? s.correctCount + 1 : s.correctCount,
          lastStudiedAt: Date.now()
        };
      }
      return s;
    }));
  };

  const handleNextQuestion = () => {
    const nextIdx = currentIndex + 1;
    if (nextIdx < activeSessionQueue.length) {
      setCurrentIndex(nextIdx);
      initQuestion(activeSessionQueue[nextIdx], studyConfig.mode);
    } else {
      // Session finished!
      finishSession();
    }
  };

  const finishSession = () => {
    const sessionRecords = records.filter(r => r.studiedAt >= sessionStartTime);
    // Calculate stats for session
    let correctCount = 0;
    let incorrectCount = 0;
    let firstTryCorrect = 0;
    const incorrectIds = new Set<string>();

    activeSessionQueue.forEach(s => {
      const sRecords = sessionRecords.filter(r => r.sentenceId === s.id);
      if (sRecords.length > 0) {
        const first = sRecords[sRecords.length - 1]; // oldest in session
        const last = sRecords[0]; // latest
        if (last.isCorrect) {
          correctCount++;
          if (first.isCorrect && first.attemptCount === 1) {
            firstTryCorrect++;
          }
        } else {
          incorrectCount++;
          incorrectIds.add(s.id);
        }
      } else {
        incorrectCount++;
        incorrectIds.add(s.id);
      }
    });

    const session: StudySession = {
      id: `ses-${Date.now()}`,
      mode: studyConfig.mode,
      totalQuestions: activeSessionQueue.length,
      correctCount,
      incorrectCount,
      firstTryCorrectCount: firstTryCorrect,
      startedAt: sessionStartTime,
      completedAt: Date.now(),
      incorrectSentenceIds: Array.from(incorrectIds)
    };

    setSessions(prev => [session, ...prev]);
    setLastCompletedSession(session);
    setCurrentView('result');
  };

  // ==========================================
  // FILTERED SENTENCES FOR LIST VIEW
  // ==========================================
  const filteredSentences = useMemo(() => {
    return sentences.filter(s => {
      const matchQuery = s.englishText.toLowerCase().includes(searchQuery.toLowerCase()) ||
                         s.koreanMeaning.toLowerCase().includes(searchQuery.toLowerCase()) ||
                         s.memo.toLowerCase().includes(searchQuery.toLowerCase());
      const matchCat = filterCategory === 'all' || s.category === filterCategory;
      const matchDiff = filterDifficulty === 'all' || s.difficulty === filterDifficulty;
      
      let matchStatus = true;
      if (filterStatus === 'favorite') matchStatus = s.isFavorite;
      else if (filterStatus === 'studied') matchStatus = s.studyCount > 0;
      else if (filterStatus === 'unstudied') matchStatus = s.studyCount === 0;

      return matchQuery && matchCat && matchDiff && matchStatus;
    }).sort((a, b) => {
      if (sortBy === 'latest') return b.createdAt - a.createdAt;
      if (sortBy === 'oldest') return a.createdAt - b.createdAt;
      if (sortBy === 'alphabetical') return a.englishText.localeCompare(b.englishText);
      return 0;
    });
  }, [sentences, searchQuery, filterCategory, filterDifficulty, filterStatus, sortBy]);

  const categories = useMemo(() => {
    const set = new Set(sentences.map(s => s.category).filter(Boolean));
    return Array.from(set);
  }, [sentences]);

  // Statistics Computations
  const statsSummary = useMemo(() => {
    const totalSentences = sentences.length;
    const totalStudyCount = sentences.reduce((acc, s) => acc + s.studyCount, 0);
    const totalCorrect = sentences.reduce((acc, s) => acc + s.correctCount, 0);
    const overallAccuracy = totalStudyCount > 0 ? Math.round((totalCorrect / totalStudyCount) * 100) : 0;
    
    // Missed sentences (lowest correct ratio)
    const missed = [...sentences]
      .filter(s => s.studyCount > 0)
      .sort((a, b) => (a.correctCount / (a.studyCount || 1)) - (b.correctCount / (b.studyCount || 1)))
      .slice(0, 5);

    return { totalSentences, totalStudyCount, overallAccuracy, missed };
  }, [sentences]);

  // Export / Import Backup JSON
  const exportData = () => {
    const backup = {
      version: STORAGE_VERSION,
      exportDate: new Date().toISOString(),
      sentences,
      records,
      sessions
    };
    const blob = new Blob([JSON.stringify(backup, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `sentenceflow_backup_${new Date().toISOString().slice(0, 10)}.json`;
    a.click();
    URL.revokeObjectURL(url);
    showToast("데이터가 JSON 파일로 백업되었습니다.");
  };

  const importData = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (event) => {
      try {
        const data = JSON.parse(event.target?.result as string);
        if (data.sentences && Array.isArray(data.sentences)) {
          setSentences(data.sentences);
          if (data.records) setRecords(data.records);
          if (data.sessions) setSessions(data.sessions);
          showToast("데이터를 성공적으로 가져왔습니다!");
        } else {
          showToast("올바르지 않은 백업 파일 형식입니다.");
        }
      } catch (err) {
        showToast("JSON 파일 파싱 중 오류가 발생했습니다.");
      }
    };
    reader.readAsText(file);
  };

  const handleFullReset = () => {
    setSentences([]);
    setRecords([]);
    setSessions([]);
    localStorage.removeItem(SENTENCES_KEY);
    localStorage.removeItem(RECORDS_KEY);
    localStorage.removeItem(SESSIONS_KEY);
    setActiveModal(null);
    showToast("모든 데이터가 초기화되었습니다.");
    setCurrentView('dashboard');
  };

  // ==========================================
  // RENDER VIEWS
  // ==========================================
  return (
    <div className="min-h-screen bg-slate-50 text-slate-800 flex flex-col font-sans antialiased selection:bg-indigo-500 selection:text-white">
      {/* Toast Notification */}
      {toastMessage && (
        <div className="fixed top-5 right-5 z-50 bg-slate-900 text-white px-5 py-3 rounded-xl shadow-xl flex items-center space-x-3 animate-fade-in border border-slate-700">
          <span className="w-2 h-2 rounded-full bg-indigo-400 animate-pulse"></span>
          <p className="text-sm font-medium">{toastMessage}</p>
        </div>
      )}

      {/* Modal Dialog */}
      {activeModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 backdrop-blur-sm p-4">
          <div className="bg-white rounded-2xl max-w-md w-full p-6 shadow-2xl border border-slate-100 animate-scale-up">
            {activeModal === 'delete' && (
              <>
                <h3 className="text-xl font-bold text-slate-900 mb-2">문장 삭제 확인</h3>
                <p className="text-slate-600 text-sm mb-6">정말 이 문장을 삭제하시겠습니까? 관련된 학습 기록은 유지되지만 목록에서 사라집니다.</p>
                <div className="flex justify-end space-x-3">
                  <button 
                    onClick={() => setActiveModal(null)} 
                    className="px-4 py-2 rounded-xl text-sm font-medium bg-slate-100 hover:bg-slate-200 text-slate-700 transition"
                  >
                    취소
                  </button>
                  <button 
                    onClick={() => handleDeleteSentence(modalData.id)} 
                    className="px-4 py-2 rounded-xl text-sm font-medium bg-rose-600 hover:bg-rose-700 text-white transition shadow-sm"
                  >
                    삭제하기
                  </button>
                </div>
              </>
            )}
            {activeModal === 'reset' && (
              <>
                <h3 className="text-xl font-bold text-rose-600 mb-2">전체 데이터 초기화</h3>
                <p className="text-slate-600 text-sm mb-6">등록된 모든 문장과 학습 기록이 영구적으로 삭제됩니다. 계속하시겠습니까?</p>
                <div className="flex justify-end space-x-3">
                  <button 
                    onClick={() => setActiveModal(null)} 
                    className="px-4 py-2 rounded-xl text-sm font-medium bg-slate-100 hover:bg-slate-200 text-slate-700 transition"
                  >
                    취소

        {currentView === 'study-setup' && (
          <div className="max-w-2xl mx-auto bg-white rounded-3xl p-6 sm:p-10 border border-slate-100 shadow-sm space-y-8 animate-fade-in">
            <div>
              <h1 className="text-2xl font-extrabold text-slate-900 mb-1">학습 설정</h1>
              <p className="text-slate-500 text-sm">원하는 학습 방식과 범위를 지정하고 집중 학습을 시작하세요.</p>
            </div>

            {/* Mode Selector */}
            <div className="space-y-3">
              <label className="text-sm font-bold text-slate-700">1. 학습 모드 선택</label>
              <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
                <button 
                  type="button"
                  onClick={() => setStudyConfig({ ...studyConfig, mode: 'word-arrange' })}
                  className={`p-4 rounded-2xl border text-left transition flex flex-col justify-between ${studyConfig.mode === 'word-arrange' ? 'border-indigo-600 bg-indigo-50/50 ring-2 ring-indigo-500/20' : 'border-slate-200 hover:bg-slate-50'}`}
                >
                  <span className="text-2xl mb-2">🧩</span>
                  <div>
                    <p className="font-bold text-slate-900 text-sm">단어 배열하기</p>
                    <p className="text-xs text-slate-500 mt-0.5">흩어진 단어를 순서대로 맞추기</p>
                  </div>
                </button>

                <button 
                  type="button"
                  onClick={() => setStudyConfig({ ...studyConfig, mode: 'typing' })}
                  className={`p-4 rounded-2xl border text-left transition flex flex-col justify-between ${studyConfig.mode === 'typing' ? 'border-indigo-600 bg-indigo-50/50 ring-2 ring-indigo-500/20' : 'border-slate-200 hover:bg-slate-50'}`}
                >
                  <span className="text-2xl mb-2">⌨️</span>
                  <div>
                    <p className="font-bold text-slate-900 text-sm">직접 입력하기</p>
                    <p className="text-xs text-slate-500 mt-0.5">영작 문장을 직접 타이핑</p>
                  </div>
                </button>

                <button 
                  type="button"
                  onClick={() => setStudyConfig({ ...studyConfig, mode: 'mixed' })}
                  className={`p-4 rounded-2xl border text-left transition flex flex-col justify-between ${studyConfig.mode === 'mixed' ? 'border-indigo-600 bg-indigo-50/50 ring-2 ring-indigo-500/20' : 'border-slate-200 hover:bg-slate-50'}`}
                >
                  <span className="text-2xl mb-2">🔄</span>
                  <div>
                    <p className="font-bold text-slate-900 text-sm">혼합 모드</p>
                    <p className="text-xs text-slate-500 mt-0.5">두 가지 방식을 번갈아 연습</p>
                  </div>
                </button>
              </div>
            </div>

            {/* Scope Selector */}
            <div className="space-y-3">
              <label className="text-sm font-bold text-slate-700">2. 학습 범위 선택</label>
              <select 
                value={studyConfig.scope}
                onChange={(e) => setStudyConfig({ ...studyConfig, scope: e.target.value as any })}
                className="w-full px-4 py-3 bg-slate-50 border border-slate-200 rounded-2xl text-sm font-medium focus:outline-none focus:border-indigo-500"
              >
                <option value="all">전체 문장 ({sentences.length}개)</option>
                <option value="favorites">즐겨찾기 문장 ({sentences.filter(s => s.isFavorite).length}개)</option>
                <option value="unstudied">아직 학습하지 않은 문장 ({sentences.filter(s => s.studyCount === 0).length}개)</option>
                <option value="incorrect">오답 복습 문장</option>
                <option value="category">특정 카테고리별</option>
              </select>

              {studyConfig.scope === 'category' && (
                <select 
                  value={studyConfig.categoryFilter}
                  onChange={(e) => setStudyConfig({ ...studyConfig, categoryFilter: e.target.value })}
                  className="w-full px-4 py-3 bg-slate-50 border border-slate-200 rounded-2xl text-sm font-medium focus:outline-none focus:border-indigo-500 mt-2"
                >
                  <option value="">카테고리 선택</option>
                  {categories.map(cat => <option key={cat} value={cat}>{cat}</option>)}
                </select>
              )}
            </div>

            {/* Question Count & Order */}
            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
              <div className="space-y-3">
                <label className="text-sm font-bold text-slate-700">3. 문제 수 설정</label>
                <select 
                  value={studyConfig.questionCount}
                  onChange={(e) => setStudyConfig({ ...studyConfig, questionCount: Number(e.target.value) })}
                  className="w-full px-4 py-3 bg-slate-50 border border-slate-200 rounded-2xl text-sm font-medium focus:outline-none focus:border-indigo-500"
                >
                  <option value={5}>5문장</option>
                  <option value={10}>10문장</option>
                  <option value={20}>20문장</option>
                  <option value={1000}>전체 문장</option>
                </select>
              </div>

              <div className="space-y-3">
                <label className="text-sm font-bold text-slate-700">4. 출제 순서</label>
                <select 
                  value={studyConfig.order}
                  onChange={(e) => setStudyConfig({ ...studyConfig, order: e.target.value as QuestionOrder })}
                  className="w-full px-4 py-3 bg-slate-50 border border-slate-200 rounded-2xl text-sm font-medium focus:outline-none focus:border-indigo-500"
                >
                  <option value="random">무작위 순서</option>
                  <option value="sequential">등록 순서</option>
                  <option value="incorrect-first">오답 우선</option>
                </select>
              </div>
            </div>

            <div className="pt-4">
              <button 
                onClick={startStudySession}
                className="w-full py-4 bg-indigo-600 hover:bg-indigo-700 text-white font-bold rounded-2xl shadow-lg shadow-indigo-200 transition transform active:scale-98"
              >
                학습 세션 시작하기 &rarr;
              </button>
            </div>
          </div>
        )}

        {/* ================= VIEW: ACTIVE STUDY SESSION ================= */}
        {currentView === 'study' && activeSessionQueue.length > 0 && (() => {
          const currentSentence = activeSessionQueue[currentIndex];
          const actualMode = studyConfig.mode === 'mixed' 
            ? ((currentIndex % 2 === 0) ? 'word-arrange' : 'typing') 
            : studyConfig.mode;
          const progressPercent = Math.round(((currentIndex) / activeSessionQueue.length) * 100);

          return (
            <div className="max-w-3xl mx-auto space-y-6 animate-fade-in">
              {/* Study Header Progress */}
              <div className="bg-white p-6 rounded-3xl border border-slate-100 shadow-sm space-y-4">
                <div className="flex items-center justify-between text-sm">
                  <span className="font-bold text-indigo-600 uppercase tracking-wider text-xs">
                    {actualMode === 'word-arrange' ? '🧩 단어 배열 모드' : '⌨️ 직접 영작 모드'}
                  </span>
                  <div className="flex items-center space-x-3">
                    <span className="font-bold text-slate-700">문제 {currentIndex + 1} / {activeSessionQueue.length}</span>
                    <button 
                      onClick={() => setCurrentView('dashboard')}
                      className="px-3 py-1 bg-slate-100 hover:bg-slate-200 text-slate-600 font-semibold rounded-xl text-xs transition"
                    >
                      학습 종료
                    </button>
                  </div>
                </div>
                <div className="w-full bg-slate-100 h-2.5 rounded-full overflow-hidden">
                  <div className="bg-indigo-600 h-full transition-all duration-300" style={{ width: `${progressPercent}%` }}></div>
                </div>
              </div>

              {/* Korean Meaning Prompt */}
              <div className="bg-white p-8 rounded-3xl border border-slate-100 shadow-sm text-center space-y-3">
                <span className="text-xs font-semibold px-3 py-1 rounded-full bg-indigo-50 text-indigo-600">{currentSentence.category || '기본'}</span>
                <h2 className="text-2xl sm:text-3xl font-extrabold text-slate-900">{currentSentence.koreanMeaning}</h2>
                {currentSentence.memo && <p className="text-xs text-slate-400">💡 힌트/메모: {currentSentence.memo}</p>}
              </div>

              {/* ================= MODE A: WORD ARRANGEMENT ================= */}
              {actualMode === 'word-arrange' && (
                <div className="bg-white p-6 sm:p-8 rounded-3xl border border-slate-100 shadow-sm space-y-6">
                  <div>
                    <label className="text-xs font-semibold text-slate-400 uppercase tracking-wider block mb-3">문장 조립 영역 (클릭하여 취소)</label>
                    <div className="min-h-[80px] p-4 bg-slate-50 border-2 border-dashed border-slate-200 rounded-2xl flex flex-wrap gap-2 items-center">
                      {assembledWords.length === 0 ? (
                        <span className="text-slate-400 text-sm italic">아래 단어 버튼을 클릭하여 문장을 완성하세요.</span>
                      ) : (
                        assembledWords.map(w => (
                          <button
                            key={w.id}
                            onClick={() => handleReturnAssembledWord(w)}
                            disabled={wordCheckResult === 'correct'}
                            className="px-4 py-2.5 bg-indigo-600 text-white font-semibold rounded-xl shadow-sm hover:bg-indigo-700 transition animate-scale-up"
                          >
                            {w.text}
                          </button>
                        ))
                      )}
                    </div>
                  </div>

                  <div>
                    <label className="text-xs font-semibold text-slate-400 uppercase tracking-wider block mb-3">사용 가능한 단어 풀</label>
                    <div className="min-h-[80px] p-4 bg-white border border-slate-200 rounded-2xl flex flex-wrap gap-2 items-center">
                      {poolWords.map(w => (
                        <button
                          key={w.id}
                          onClick={() => handleSelectPoolWord(w)}
                          disabled={wordCheckResult === 'correct'}
                          className="px-4 py-2.5 bg-slate-100 hover:bg-indigo-50 hover:border-indigo-300 hover:text-indigo-700 text-slate-700 font-semibold rounded-xl border border-slate-200 transition shadow-xs"
                        >
                          {w.text}
                        </button>
                      ))}
                    </div>
                  </div>

                  {/* Feedback or Actions */}
                  {wordCheckResult === 'correct' ? (
                    <div className="p-6 bg-emerald-50 border border-emerald-200 rounded-2xl text-center space-y-4 animate-fade-in">
                      <div className="flex items-center justify-center space-x-2 text-emerald-700 font-bold text-lg">
                        <span>🎉 정답입니다! 완벽해요.</span>
                      </div>
                      <button 
                        onClick={handleNextQuestion}
                        className="w-full py-3.5 bg-emerald-600 hover:bg-emerald-700 text-white font-bold rounded-xl shadow-md transition"
                      >
                        다음 문제로 &rarr;
                      </button>
                    </div>
                  ) : wordCheckResult === 'incorrect' ? (
                    <div className="p-6 bg-rose-50 border border-rose-200 rounded-2xl text-center space-y-4 animate-fade-in">
                      <div className="space-y-1">
                        <p className="text-rose-700 font-bold text-base">⚠️ 아쉽네요, 정답이 아닙니다.</p>
                        <p className="text-xs text-slate-600">정답: <strong className="text-slate-900">{currentSentence.englishText}</strong></p>
                      </div>
                      <div className="flex space-x-3">
                        <button 
                          onClick={handleResetWordArrange}
                          className="flex-1 py-3 bg-white border border-rose-200 text-rose-700 font-semibold rounded-xl hover:bg-rose-100 transition text-sm"
                        >
                          다시 시도하기
                        </button>
                        <button 
                          onClick={handleNextQuestion}
                          className="flex-1 py-3 bg-rose-600 text-white font-semibold rounded-xl hover:bg-rose-700 transition text-sm"
                        >
                          다음 문제로 넘어가기
                        </button>
                      </div>
                    </div>
                  ) : (
                    <div className="flex items-center justify-between pt-2">
                      <button 
                        onClick={handleResetWordArrange}
                        className="px-5 py-3 bg-slate-100 hover:bg-slate-200 text-slate-700 font-semibold rounded-2xl text-sm transition"
                      >
                        처음부터 다시
                      </button>
                      <button 
                        onClick={handleCheckWordArrange}
                        className="px-8 py-3 bg-indigo-600 hover:bg-indigo-700 text-white font-bold rounded-2xl shadow-md transition"
                      >
                        정답 확인
                      </button>
                    </div>
                  )}
                </div>
              )}

              {/* ================= MODE B: DIRECT TYPING ================= */}
              {actualMode === 'typing' && (
                <div className="bg-white p-6 sm:p-8 rounded-3xl border border-slate-100 shadow-sm space-y-6">
                  <div>
                    <label className="text-xs font-semibold text-slate-400 uppercase tracking-wider block mb-3">영어로 문장을 직접 입력하세요</label>
                    <textarea 
                      rows={3}
                      value={typedAnswer}
                      onChange={(e) => setTypedAnswer(e.target.value)}
                      disabled={typingCheckResult === 'correct'}
                      placeholder="Type the English sentence here..."
                      onKeyDown={(e) => {
                        if (e.key === 'Enter' && !e.shiftKey) {
                          e.preventDefault();
                          if (typingCheckResult !== 'correct') handleCheckTyping();
                        }
                      }}
                      className="w-full p-4 bg-slate-50 border border-slate-200 rounded-2xl text-base focus:outline-none focus:border-indigo-500 transition resize-none"
                    />
                  </div>

                  {/* Feedback or Actions */}
                  {typingCheckResult === 'correct' ? (
                    <div className="p-6 bg-emerald-50 border border-emerald-200 rounded-2xl text-center space-y-4 animate-fade-in">
                      <div className="flex items-center justify-center space-x-2 text-emerald-700 font-bold text-lg">
                        <span>🎉 정답입니다! 영작 실력이 훌륭해요.</span>
                      </div>
                      <button 
                        onClick={handleNextQuestion}
                        className="w-full py-3.5 bg-emerald-600 hover:bg-emerald-700 text-white font-bold rounded-xl shadow-md transition"
                      >
                        다음 문제로 &rarr;
                      </button>
                    </div>
                  ) : typingCheckResult === 'incorrect' ? (
                    <div className="p-6 bg-rose-50 border border-rose-200 rounded-2xl text-center space-y-4 animate-fade-in">
                      <div className="space-y-2 text-left">
                        <p className="text-rose-700 font-bold text-base text-center">⚠️ 오답입니다. 비교하여 차이점을 확인하세요.</p>
                        <div className="p-3 bg-white rounded-xl border border-rose-200 text-xs space-y-1">
                          <p><span className="font-semibold text-slate-500">내가 입력한 답:</span> <span className="text-rose-600">{typedAnswer}</span></p>
                          <p><span className="font-semibold text-slate-500">정답 문장:</span> <span className="text-emerald-600 font-bold">{currentSentence.englishText}</span></p>
                        </div>
                      </div>
                      <div className="flex space-x-3">
                        <button 
                          onClick={() => setTypingCheckResult('idle')}
                          className="flex-1 py-3 bg-white border border-rose-200 text-rose-700 font-semibold rounded-xl hover:bg-rose-100 transition text-sm"
                        >
                          다시 입력하기
                        </button>
                        <button 
                          onClick={handleNextQuestion}
                          className="flex-1 py-3 bg-rose-600 text-white font-semibold rounded-xl hover:bg-rose-700 transition text-sm"
                        >
                          다음 문제로
                        </button>
                      </div>
                    </div>
                  ) : (
                    <div className="flex items-center justify-between pt-2">
                      <button 
                        onClick={() => setShowAnswerInTyping(!showAnswerInTyping)}
                        className="px-5 py-3 bg-amber-50 hover:bg-amber-100 text-amber-700 font-semibold rounded-2xl text-sm transition"
                      >
                        {showAnswerInTyping ? '정답 숨기기' : '💡 정답 보기'}
                      </button>
                      <button 
                        onClick={handleCheckTyping}
                        className="px-8 py-3 bg-indigo-600 hover:bg-indigo-700 text-white font-bold rounded-2xl shadow-md transition"
                      >
                        정답 제출
                      </button>
                    </div>
                  )}

                  {showAnswerInTyping && typingCheckResult !== 'correct' && (
                    <div className="p-4 bg-amber-50/60 border border-amber-200 rounded-2xl text-amber-900 text-sm animate-fade-in">
                      <p className="font-bold mb-1">정답 힌트:</p>
                      <p className="font-mono text-base">{currentSentence.englishText}</p>
                    </div>
                  )}
                </div>
              )}
            </div>
          );
        })()}

        {/* ================= VIEW: STUDY RESULT ================= */}
        {currentView === 'result' && lastCompletedSession && (
          <div className="max-w-2xl mx-auto bg-white rounded-3xl p-6 sm:p-10 border border-slate-100 shadow-sm space-y-8 animate-fade-in">
            <div className="text-center space-y-3">
              <div className="w-16 h-16 bg-indigo-100 text-indigo-600 rounded-3xl mx-auto flex items-center justify-center text-3xl font-bold shadow-md">
                🏆
              </div>
              <h1 className="text-2xl sm:text-3xl font-extrabold text-slate-950">학습 세션 완료!</h1>
              <p className="text-slate-500 text-sm">오늘도 꾸준한 학습으로 영어 실력이 향상되었습니다.</p>
            </div>

            {/* Results Grid */}
            <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
              <div className="bg-slate-50 p-4 rounded-2xl border border-slate-100 text-center">
                <span className="text-xs text-slate-400 font-semibold uppercase block mb-1">총 문제 수</span>
                <span className="text-2xl font-extrabold text-slate-900">{lastCompletedSession.totalQuestions}</span>
              </div>
              <div className="bg-slate-50 p-4 rounded-2xl border border-slate-100 text-center">
                <span className="text-xs text-slate-400 font-semibold uppercase block mb-1">정답 수</span>
                <span className="text-2xl font-extrabold text-emerald-600">{lastCompletedSession.correctCount}</span>
              </div>
              <div className="bg-slate-50 p-4 rounded-2xl border border-slate-100 text-center">
                <span className="text-xs text-slate-400 font-semibold uppercase block mb-1">첫 시도 정답</span>
                <span className="text-2xl font-extrabold text-indigo-600">{lastCompletedSession.firstTryCorrectCount}</span>
              </div>
              <div className="bg-slate-50 p-4 rounded-2xl border border-slate-100 text-center">
                <span className="text-xs text-slate-400 font-semibold uppercase block mb-1">정답률</span>
                <span className="text-2xl font-extrabold text-slate-900">
                  {Math.round((lastCompletedSession.correctCount / lastCompletedSession.totalQuestions) * 100)}%
                </span>
              </div>
            </div>

            {/* Incorrect Sentences Review */}
            {lastCompletedSession.incorrectSentenceIds.length > 0 && (
              <div className="space-y-3">
                <h3 className="text-sm font-bold text-slate-700">이번에 틀린 문장 복습</h3>
                <div className="space-y-2 max-h-60 overflow-y-auto">
                  {lastCompletedSession.incorrectSentenceIds.map(id => {
                    const s = sentences.find(item => item.id === id);
                    if (!s) return null;
                    return (
                      <div key={id} className="p-3 bg-rose-50/50 border border-rose-100 rounded-2xl flex items-center justify-between text-sm">
                        <div>
                          <p className="font-bold text-slate-800">{s.englishText}</p>
                          <p className="text-xs text-slate-500">{s.koreanMeaning}</p>
                        </div>
                      </div>
                    );
                  })}
                </div>
              </div>
            )}

            {/* Actions */}
            <div className="flex flex-col sm:flex-row gap-3 pt-4">
              <button 
                onClick={() => setCurrentView('study-setup')}
                className="flex-1 py-3.5 bg-indigo-600 hover:bg-indigo-700 text-white font-bold rounded-2xl shadow-md transition text-center"
              >
                다시 학습하기
              </button>
              <button 
                onClick={() => setCurrentView('dashboard')}
                className="flex-1 py-3.5 bg-slate-100 hover:bg-slate-200 text-slate-700 font-semibold rounded-2xl transition text-center"
              >
                대시보드로 돌아가기
              </button>
            </div>
          </div>
        )}

        {/* ================= VIEW: STATISTICS ================= */}
        {currentView === 'stats' && (
          <div className="space-y-8 animate-fade-in">
            <div>
              <h1 className="text-2xl font-extrabold text-slate-900">학습 통계</h1>
              <p className="text-slate-500 text-sm">나의 누적 학습 데이터와 취약한 문장을 분석합니다.</p>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
              <div className="bg-white p-6 rounded-3xl border border-slate-100 shadow-sm">
                <span className="text-xs font-semibold text-slate-400 uppercase tracking-wider block mb-2">총 학습 완료 세션</span>
                <span className="text-4xl font-extrabold text-slate-900">{sessions.length}</span>
                <span className="text-xs text-slate-500 ml-1">회</span>
              </div>

              <div className="bg-white p-6 rounded-3xl border border-slate-100 shadow-sm">
                <span className="text-xs font-semibold text-slate-400 uppercase tracking-wider block mb-2">누적 정답 수</span>
                <span className="text-4xl font-extrabold text-emerald-600">
                  {sentences.reduce((acc, s) => acc + s.correctCount, 0)}
                </span>
                <span className="text-xs text-slate-500 ml-1">개</span>
              </div>

              <div className="bg-white p-6 rounded-3xl border border-slate-100 shadow-sm">
                <span className="text-xs font-semibold text-slate-400 uppercase tracking-wider block mb-2">전체 평균 정답률</span>
                <span className="text-4xl font-extrabold text-indigo-600">{statsSummary.overallAccuracy}%</span>
              </div>
            </div>

            {/* Frequently Missed Sentences */}
            <div className="bg-white rounded-3xl p-6 sm:p-8 border border-slate-100 shadow-sm space-y-4">
              <h2 className="text-lg font-bold text-slate-900">자주 틀리는 취약 문장 Top 5</h2>
              {statsSummary.missed.length === 0 ? (
                <p className="text-slate-400 text-sm py-6 text-center">아직 학습 기록이 충분하지 않습니다.</p>
              ) : (
                <div className="space-y-3">
                  {statsSummary.missed.map(s => {
                    const acc = s.studyCount > 0 ? Math.round((s.correctCount / s.studyCount) * 100) : 0;
                    return (
                      <div key={s.id} className="p-4 rounded-2xl bg-slate-50 border border-slate-100 flex items-center justify-between">
                        <div className="space-y-1">
                          <p className="font-bold text-slate-800 text-sm">{s.englishText}</p>
                          <p className="text-xs text-slate-500">{s.koreanMeaning}</p>
                        </div>
                        <div className="text-right">
                          <span className="text-sm font-extrabold text-rose-600">{acc}%</span>
                          <p className="text-[10px] text-slate-400">정답률</p>
                        </div>
                      </div>
                    );
                  })}
                </div>
              )}
            </div>
          </div>
        )}

      </main>

      {/* Footer */}
      <footer className="bg-white border-t border-slate-200 py-6 mt-auto text-center text-xs text-slate-400">
        <p>© 2026 SentenceFlow. Designed for ultimate English mastery.</p>
      </footer>
    </div>
  );
}

// ==========================================
// SUB-COMPONENT: SENTENCE FORM VIEW
// ==========================================
interface SentenceFormViewProps {
  sentenceToEdit?: Sentence;
  onSave: (data: {
    englishText: string;
    koreanMeaning: string;
    category: string;
    difficulty: Difficulty;
    memo: string;
    isFavorite: boolean;
  }) => void;
  onCancel: () => void;
}

function SentenceFormView({ sentenceToEdit, onSave, onCancel }: SentenceFormViewProps) {
  const [englishText, setEnglishText] = useState(sentenceToEdit?.englishText || '');
  const [koreanMeaning, setKoreanMeaning] = useState(sentenceToEdit?.koreanMeaning || '');
  const [category, setCategory] = useState(sentenceToEdit?.category || '기초');
  const [difficulty, setDifficulty] = useState<Difficulty>(sentenceToEdit?.difficulty || 'Easy');
  const [memo, setMemo] = useState(sentenceToEdit?.memo || '');
  const [isFavorite, setIsFavorite] = useState(sentenceToEdit?.isFavorite || false);

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    onSave({
      englishText,
      koreanMeaning,
      category,
      difficulty,
      memo,
      isFavorite
    });
  };

  return (
    <div className="max-w-2xl mx-auto bg-white rounded-3xl p-6 sm:p-10 border border-slate-100 shadow-sm space-y-6 animate-fade-in">
      <div>
        <h1 className="text-2xl font-extrabold text-slate-900 mb-1">
          {sentenceToEdit ? '문장 수정하기' : '새 영어 문장 등록'}
        </h1>
        <p className="text-slate-500 text-sm">학습하고 싶은 영어 문장과 뜻을 입력하세요.</p>
      </div>

      <form onSubmit={handleSubmit} className="space-y-5">
        <div>
          <label className="block text-xs font-semibold text-slate-700 uppercase tracking-wider mb-2">영어 문장 <span className="text-rose-500">*</span></label>
          <textarea 
            rows={3}
            required
            value={englishText}
            onChange={(e) => setEnglishText(e.target.value)}
            placeholder="e.g. I want to learn English fluently."
            className="w-full p-4 bg-slate-50 border border-slate-200 rounded-2xl text-sm focus:outline-none focus:border-indigo-500 transition resize-none font-medium"
          />
        </div>

        <div>
          <label className="block text-xs font-semibold text-slate-700 uppercase tracking-wider mb-2">한국어 뜻 <span className="text-rose-500">*</span></label>
          <input 
            type="text"
            required
            value={koreanMeaning}
            onChange={(e) => setKoreanMeaning(e.target.value)}
            placeholder="e.g. 나는 영어를 유창하게 배우고 싶다."
            className="w-full px-4 py-3.5 bg-slate-50 border border-slate-200 rounded-2xl text-sm focus:outline-none focus:border-indigo-500 transition font-medium"
          />
        </div>

        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
          <div>
            <label className="block text-xs font-semibold text-slate-700 uppercase tracking-wider mb-2">카테고리</label>
            <input 
              type="text"
              value={category}
              onChange={(e) => setCategory(e.target.value)}
              placeholder="e.g. 비즈니스, 일상, 여행"
              className="w-full px-4 py-3 bg-slate-50 border border-slate-200 rounded-2xl text-sm focus:outline-none focus:border-indigo-500 transition"
            />
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-700 uppercase tracking-wider mb-2">난이도</label>
            <select 
              value={difficulty}
              onChange={(e) => setDifficulty(e.target.value as Difficulty)}
              className="w-full px-4 py-3 bg-slate-50 border border-slate-200 rounded-2xl text-sm font-medium focus:outline-none focus:border-indigo-500"
            >
              <option value="Easy">Easy (쉬움)</option>
              <option value="Medium">Medium (보통)</option>
              <option value="Hard">Hard (어려움)</option>
            </select>
          </div>
        </div>

        <div>
          <label className="block text-xs font-semibold text-slate-700 uppercase tracking-wider mb-2">메모 / 문법 팁 (선택)</label>
          <input 
            type="text"
            value={memo}
            onChange={(e) => setMemo(e.target.value)}
            placeholder="e.g. want to + 동사원형 패턴"
            className="w-full px-4 py-3 bg-slate-50 border border-slate-200 rounded-2xl text-sm focus:outline-none focus:border-indigo-500 transition"
          />
        </div>

        <div className="flex items-center space-x-3 pt-2">
          <input 
            type="checkbox"
            id="favoriteCheckbox"
            checked={isFavorite}
            onChange={(e) => setIsFavorite(e.target.checked)}
            className="w-5 h-5 rounded-lg text-indigo-600 focus:ring-indigo-500 border-slate-300"
          />
          <label htmlFor="favoriteCheckbox" className="text-sm font-semibold text-slate-700 cursor-pointer">
            즐겨찾기에 추가 ⭐
          </label>
        </div>

        <div className="flex space-x-3 pt-4">
          <button 
            type="button"
            onClick={onCancel}
            className="flex-1 py-3.5 bg-slate-100 hover:bg-slate-200 text-slate-700 font-semibold rounded-2xl transition text-sm"
          >
            취소
          </button>
          <button 
            type="submit"
            className="flex-1 py-3.5 bg-indigo-600 hover:bg-indigo-700 text-white font-bold rounded-2xl shadow-md transition text-sm"
          >
            {sentenceToEdit ? '수정 완료' : '저장하기'}
          </button>
        </div>
      </form>
    </div>
  );
}
