let windPower=0.55;

function randomWind(){

    const difficulty =
    parseFloat(
        document.getElementById(
            "difficulty"
        ).value
    );

    windPower =
    Math.random()*difficulty*8;

    windAngle =
    Math.random()*Math.PI*2;


    const dirs=[
        "→",
        "↗",
        "↑",
        "↖",
        "←",
        "↙",
        "↓",
        "↘"
    ];

    let index =
    Math.round(
        windAngle/(Math.PI/4)
    )%8;

    document.getElementById(
        "windArrow"
    ).textContent=
    dirs[index];

    document.getElementById(
        "windPower"
    ).textContent=
    windPower.toFixed(1)+" m/s";
}

function shoot(){

    if(
        arrowFlying ||
        gameFinished
    ) return;


    const dx =
    aimCircle.position.x -
    launchCircle.position.x;

    const dy =
    aimCircle.position.y -
    launchCircle.position.y;


    const distance =
    Math.sqrt(
        dx*dx+
        dy*dy
    );


    if(distance > .65){

        document.getElementById(
            "lock"
        ).textContent=
        "조준점을 맞춰주세요!";

        return;
    }


    locked=true;

    document.getElementById(
        "shootButton"
    ).disabled=true;


    arrowMesh =
    createArrow();

    arrowMesh.position.set(
        0,
        3,
        12
    );

    scene.add(arrowMesh);


    arrowFlying=true;

    arrowMesh.userData={
        t:0,
        x:0,
        y:3,
        z:12
    };


    /* 활시위 */

    string.visible=false;
}

let sensorEnabled=false;


function handleOrientation(e){

    if(
        e.gamma===null ||
        e.beta===null
    ) return;


    aimX =
    THREE.MathUtils.clamp(
        e.gamma/25,
        -1,
        1
    )*4;


    aimY =
    THREE.MathUtils.clamp(
        (e.beta-45)/25,
        -1,
        1
    )*3;

}


/* iOS 센서 권한 */

async function enableSensor(){

    if(
        typeof DeviceOrientationEvent !==
        "undefined" &&
        typeof DeviceOrientationEvent.requestPermission ===
        "function"
    ){

        try{

            const permission =
            await DeviceOrientationEvent
            .requestPermission();

            if(
                permission==="granted"
            ){

                window.addEventListener(
                    "deviceorientation",
                    handleOrientation
                );

                sensorEnabled=true;
            }

        }catch(err){

            console.log(err);

        }

    }else{

        window.addEventListener(
            "deviceorientation",
            handleOrientation
        );

        sensorEnabled=true;
    }

}


window.addEventListener(
    "touchstart",
    ()=>{
        if(!sensorEnabled)
            enableSensor();
    },
    {once:true}
);


/* =====================================================
   점수 계산
===================================================== */

function calculateScore(){

    const hitX =
    arrowMesh.position.x -
    target.position.x;

    const hitY =
    arrowMesh.position.y -
    target.position.y;


    const distance =
    Math.sqrt(
        hitX*hitX+
        hitY*hitY
    );


    let points=0;


    if(distance < .8)
        points=100;

    else if(distance < 1.7)
        points=80;

    else if(distance < 2.5)
        points=60;

    else if(distance < 3.2)
        points=40;

    else if(distance < 4.2)
        points=20;


    return points;
}

   명중

    score += points;


    document.getElementById(

    gameFinished=true;


    setTimeout(()=>{

        document.getElementById(
            "finalScore"
        ).textContent=

        document.getElementById(

    },700);

}

/* =====================================================
   게임 업데이트
===================================================== */

function updateArrow(){

    if(
        !arrowFlying ||
        !arrowMesh
    ) return;


    const data =
    arrowMesh.userData;


    data.t += .018;


    /* 기본 비행 */

    data.z -= .55;


    /* 조준 방향 */

    data.x +=
    (aimX-data.x)*.015;


    data.y +=
    (3+aimY-data.y)*.015;


    /* 바람 */

    const windX =
    Math.cos(windAngle)*
    windPower*.002;

    const windY =
    Math.sin(windAngle)*
    windPower*.002;


    data.x += windX;
    data.y += windY;


    arrowMesh.position.set(
        data.x,
        data.y,
        data.z
    );


    /* 화살 방향 */

    arrowMesh.lookAt(
        data.x,
        data.y,
        data.z-2
    );


    /* 과녁 근처 */

    if(
        data.z <=
        target.position.z
    ){

        arrowFlying=false;


        arrowMesh.position.z=
        target.position.z-.7;


        const points =
        calculateScore();


        if(points>0){

            finishGame();

        }else{

            document.getElementById(
                "lock"
            ).textContent=
            "빗나갔습니다!";

            setTimeout(()=>{

                location.reload();

            },1200);

        }

    }

}


/* =====================================================
   조준 원 업데이트
===================================================== */

function updateAim(){

    if(arrowFlying)
        return;


    aimCircle.position.x =
    aimX;


    aimCircle.position.y =
    3+aimY;


    const distance =
    aimCircle.position.distanceTo(
        launchCircle.position
    );


    if(distance<.65){

        locked=true;

        document.getElementById(
            "lock"
        ).textContent=
        "✓ LOCKED";

        document.getElementById(
            "lock"
        ).classList.add(
            "ready"
        );

    }else{

        locked=false;

        document.getElementById(
            "lock"
        ).textContent=
        "조준 원을 발사 원에 맞추세요";

        document.getElementById(
            "lock"
        ).classList.remove(
            "ready"
        );

    }

}
